\begin{frame}{A four-vertex asymmetric instance}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \centering
      \theyebrow{unit-arc shortest-path metric}

      \smallskip
      \resizebox{0.82\linewidth}{!}{\ThAsymFourVertexSeed}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.64em}
        \item The root $r=a$ and vertices $b,c$ are always active.
        \item Vertex $d$ is active independently with probability $1/2$.
        \item The arc $b\to c$ remains direct, while its reverse is
          replaced by $c\to d\to b$.
      \end{itemize}
    \end{column}
  \end{columns}
  \thsource{Thesis, Section 5.1}
\end{frame}

\begin{frame}{Activation reverses the best tour}
  \centering
  \resizebox{0.76\linewidth}{!}{\ThAsymFourVertexOutcomes}

  \vspace{0.10cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
      \small
      An adaptive policy chooses its first probe before learning whether
      $d$ is active, so it cannot realize both highlighted optima. Probing
      $c,d,b$ costs $4$ in either realization. The fixed order
      $a,c,d,b,a$ does as well.
    \end{column}
    \begin{column}{0.34\textwidth}
      \vspace{-0.15cm}
      \[
        \optpost=\frac72,\qquad
        \optadapt=\optprior=4.
      \]
      \vspace{-0.28cm}
      \[
        \operatorname{CG}=\frac{\optadapt}{\optpost}=\frac87.
      \]
    \end{column}
  \end{columns}
  \thsource{Thesis, Section 5.1}
\end{frame}
